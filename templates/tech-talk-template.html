Before & After

Before


// Old approach
function fetch(url) {
  var xhr = new XMLHttpRequest();
  xhr.open('GET', url);
  xhr.onload = function() {
    callback(xhr.response);
  };
  xhr.send();
}

After


// Modern approach
async function fetch(url) {
  const response = await fetch(url);
  return response.json();
}
